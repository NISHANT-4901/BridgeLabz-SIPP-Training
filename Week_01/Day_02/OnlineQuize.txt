class QuizTimer {
    Attribute timeSlots  
    Attribute totalQuestions
    Attribute currentIndex

    Constructor(timeSlotsArray){
        set timeSlots = timeSlotsArray
        set totalQuestions = length of timeSlotsArray
        set currentIndex = 0
    }

    Method startQuiz(){
        for q = 1 to totalQuestions:
            set allottedTime = timeSlots[currentIndex]

            Print "Question " + q + " has " + allottedTime + " seconds."

            for t = allottedTime down to 1:
                if t <= 3 then
                    Print "Warning: Only " + t + " seconds left!"
                else
                    Print "Time remaining: " + t + " seconds"

            Print "Time over for Question " + q
            moveToNext()
    }

    Method moveToNext(){
        currentIndex = (currentIndex + 1) mod totalQuestions
    }
}
