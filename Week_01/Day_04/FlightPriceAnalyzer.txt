class Flight {
    Attribute flightNumber
    Attribute price
    Attribute duration

    Constructor(flightNumber, price, duration) {
        set flightNumber = flightNumber
        set price = price
        set duration = duration
    }
}

class FlightPriceDurationComparator {
    Method compare(flight1, flight2) {
        if flight1.price < flight2.price then
            return -1
        else if flight1.price > flight2.price then
            return 1
        else
            if flight1.duration < flight2.duration then
                return -1
            else if flight1.duration > flight2.duration then
                return 1
            else
                return 0
    }
}

class FlightPriceAnalyzer {
    Method mergeSort(flights, comparator) {
        if length of flights <= 1 then
            return flights

        mid = length of flights / 2
        left = mergeSort(flights[0 to mid-1], comparator)
        right = mergeSort(flights[mid to end], comparator)

        return merge(left, right, comparator)
    }

    Method merge(left, right, comparator) {
        result = new List()
        i = 0
        j = 0

        while i < length of left and j < length of right do
            if comparator.compare(left[i], right[j]) <= 0 then
                append left[i] to result
                i = i + 1
            else
                append right[j] to result
                j = j + 1

        while i < length of left do
            append left[i] to result
            i = i + 1

        while j < length of right do
            append right[j] to result
            j = j + 1

        return result
    }
}

flights = [
    new Flight("F101", 300, 180),
    new Flight("F102", 250, 200),
    new Flight("F103", 300, 150),
    new Flight("F104", 400, 120),
    new Flight("F105", 250, 190)
]

comparator = new FlightPriceDurationComparator()
analyzer = new FlightPriceAnalyzer()
sortedFlights = analyzer.mergeSort(flights, comparator)

for each flight in sortedFlights do
    Print flight.flightNumber + " Price: " + flight.price + " Duration: " + flight.duration
