class Customer {
    Attribute name
    Attribute address

    Constructor(name, address){
        set name = name
        set address = address
    }

    Method show(){
        Print "Customer: " + name + ", Address: " + address
    }
}

class Restaurant {
    Attribute name
    Attribute location

    Constructor(name, location){
        set name = name
        set location = location
    }

    Method show(){
        Print "Restaurant: " + name + ", Location: " + location
    }
}

class DeliveryBoy {
    Attribute name
    Attribute contact

    Constructor(name, contact){
        set name = name
        set contact = contact
    }

    Method show(){
        Print "Delivery Boy: " + name + ", Contact: " + contact
    }
}

class Order {
    Attribute customer       // composition
    Attribute restaurant     // composition
    Attribute deliveryBoy    // composition
    Attribute items          // List<String>
    Attribute totalPrice

    Constructor(customer, restaurant, deliveryBoy, items, totalPrice){
        set this.customer = customer
        set this.restaurant = restaurant
        set this.deliveryBoy = deliveryBoy
        set this.items = items
        set this.totalPrice = totalPrice
    }

    Method showOrder(){
        Print "--- Order Details ---"
        customer.show()
        restaurant.show()
        deliveryBoy.show()
        Print "Items: " + items
        Print "Total Price: " + totalPrice
    }
}

class Main {
    Method run(){
        cust = new Customer("Rahul", "MG Road, Delhi")
        rest = new Restaurant("Pizza Hub", "Connaught Place")
        boy = new DeliveryBoy("Amit", "9876543210")

        itemList = ["Pizza", "Coke"]
        order = new Order(cust, rest, boy, itemList, 550)

        order.showOrder()
    }
}
