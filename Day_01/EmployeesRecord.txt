class Employee {
    Attribute id
    Attribute name
    Attribute department

    Constructor(id, name, department){
        set id = id
        set name = name
        set department = department
    }

    Method show(){
        Print id
        Print name
        Print department
    }
}

class EmployeeRecord {
    Attribute employees   // ArrayList<Employee>

    Constructor(){
        employees = new ArrayList()
    }

    Method addEmployee(emp){
        employees.add(emp)
    }

    Method showAllEmployees(){
        For each emp in employees:
            emp.show()
    }

    Method filterByDepartment(deptName){
        filteredList = employees.stream()
                                .filter(emp -> emp.department == deptName)
                                .collect(toList)

        Print "Employees in Department: " + deptName
        For each emp in filteredList:
            emp.show()
    }
}
