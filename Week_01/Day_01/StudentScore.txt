class StudentScores {
    Attribute scores   // HashMap<String, Integer>

    Constructor(){
        scores = new HashMap()
    }

    Method addScore(name, marks){
        scores.put(name, marks)
    }

    Method showScores(){
        For each (name, marks) in scores:
            Print name + " : " + marks
    }

    Method getTopScorer(){
        topEntry = scores.entrySet()
                         .stream()
                         .max(compare by marks)
                         .get()

        Print "Top Scorer: " + topEntry.key + " with " + topEntry.value + " marks"
    }
}
