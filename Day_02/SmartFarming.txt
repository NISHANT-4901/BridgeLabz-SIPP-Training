abstract class Crop {
    Attribute name

    Constructor(name){
        set name = name
    }

    Abstract Method waterRequirement()
    Abstract Method harvest()
}

class Rice extend Crop {
    Constructor(){
        super("Rice")
    }

    Method waterRequirement(){
        Print "Rice needs high water supply (about 1200mm)"
    }

    Method harvest(){
        Print "Rice harvested after 4–5 months"
    }
}

class Wheat extend Crop {
    Constructor(){
        super("Wheat")
    }

    Method waterRequirement(){
        Print "Wheat needs moderate water supply (about 450mm)"
    }

    Method harvest(){
        Print "Wheat harvested after 3–4 months"
    }
}

class SmartFarmingSystem {
    Method run(){
        Crop c1 = new Rice()
        Crop c2 = new Wheat()

        c1.waterRequirement()
        c1.harvest()

        c2.waterRequirement()
        c2.harvest()
    }
}
