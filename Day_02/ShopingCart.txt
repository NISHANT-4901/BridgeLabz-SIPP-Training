class Item {
    Attribute name
    Attribute price
    Attribute quantity

    Constructor(name, price, quantity){
        set name = name
        set price = price
        set quantity = quantity
    }

    Method show(){
        Print name + " | Price: " + price + " | Qty: " + quantity
    }

    Method getTotal(){
        return price * quantity
    }
}

class ShoppingCart {
    Attribute cart   // List<Item>

    Constructor(){
        cart = new List()
    }

    Method addItem(item){
        cart.add(item)
    }

    Method showCart(){
        For each item in cart:
            item.show()
    }

    Generic Method calculateTotal(List<T> items){
        total = 0
        For each element in items:
            total = total + element.getTotal()
        return total
    }
}

class Main {
    Method run(){
        cart = new ShoppingCart()

        item1 = new Item("Apple", 10, 3)
        item2 = new Item("Milk", 50, 2)

        cart.addItem(item1)
        cart.addItem(item2)

        cart.showCart()

        totalPrice = cart.calculateTotal(cart.cart)
        Print "Total Price: " + totalPrice
    }
}
