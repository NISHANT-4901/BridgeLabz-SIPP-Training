class Job {
    Attribute id
    Attribute deadline     // time by which job must finish
    Attribute processingTime  // time required to complete job

    Constructor(id, deadline, processingTime){
        set id = id
        set deadline = deadline
        set processingTime = processingTime
    }

    Method show(){
        Print "Job ID: " + id + 
              ", Deadline: " + deadline + 
              ", Processing Time: " + processingTime
    }
}

class JobScheduler {
    Attribute jobQueue   // PriorityQueue<Job>

    Constructor(){
        // Jobs with earlier deadline get higher priority
        // If deadlines are same → shorter processing time first
        jobQueue = new PriorityQueue(
            (job1, job2) -> 
                if job1.deadline != job2.deadline then
                    return job1.deadline - job2.deadline
                else
                    return job1.processingTime - job2.processingTime
        )
    }

    Method addJob(job){
        jobQueue.add(job)
    }

    Method scheduleJobs(){
        while jobQueue is not empty:
            job = jobQueue.poll()   // get highest priority job
            Print "Executing: "
            job.show()
    }
}
