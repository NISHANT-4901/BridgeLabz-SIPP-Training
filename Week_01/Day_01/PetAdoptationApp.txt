class Pet {
    Attribute name 
    Attribute age 

    Constructor(name, age){
        set name = name 
        set age = age
    }

    Method show(){
        Print name 
        Print age 
    }
}

class Dog extend Pet {
    Attribute isFriendly 

    Constructor(name, age, isFriendly){
        super(name, age) 
        set isFriendly = isFriendly
    }

    Method show(){
        super show()
        Print isFriendly
    }
}
